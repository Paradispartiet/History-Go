import fs from 'node:fs';
import path from 'node:path';

const ROLE='subkultur_produksjon_og_prosjekt';
const KEY=`subkultur/${ROLE}`;
const WORLD=`data/Civication/roleWorlds/subkultur/${ROLE}.json`;
const MODEL=`data/Civication/roleModels/subkultur/${ROLE}.json`;
const GRAMMAR=`data/Civication/workGrammars/subkultur/${ROLE}.json`;
const PLAN=`data/Civication/mailPlans/subkultur/${ROLE}_plan.json`;
const TYPES=['job','people','conflict','story','event','micro','followup','knowledge','consequence'];
const writeJson=(p,o)=>{fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,JSON.stringify(o,null,2)+'\n');};
const writeText=(p,s)=>{fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,s.trimEnd()+'\n');};
const read=p=>JSON.parse(fs.readFileSync(p,'utf8'));

const grammar=read(GRAMMAR), model=read(MODEL), plan=read(PLAN);
if(plan.sequence.length!==16) throw new Error('expected 16-step prerequisite plan');
if(grammar.persistent_work_object_contract?.id!=='prosjektplan_budsjett_kontrakt_og_avvikslogg') throw new Error('persistent work object mismatch');
if((model.related_people||[]).length!==4) throw new Error('expected four prerequisite work actors');

const sourceRefs=[];
for(const type of TYPES){
  const f=`data/Civication/mailFamilies/subkultur/${type}/${ROLE}_${type}.json`;
  const c=read(f);
  for(const m of (c.families||[]).flatMap(x=>x.mails||[])) sourceRefs.push(`${f}#${m.id}`);
}
if(sourceRefs.length!==15||new Set(sourceRefs).size!==15) throw new Error(`expected 15 unique source refs, got ${sourceRefs.length}`);
